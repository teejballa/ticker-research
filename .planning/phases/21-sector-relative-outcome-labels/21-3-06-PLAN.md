```typescript
export function classifyHit(args: {
  ticker_return_pct: number;
  spy_return_pct: number | null;
  sector_relative_pct?: number | null;   // NEW — primary path when provided
  sector_sigma?: number | null;          // NEW (WARNING-1) — optional sector volatility (e.g. 60d rolling stdev of sector ETF daily returns)
  k?: number;                            // NEW (WARNING-1) — multiplier for σ-driven threshold; default 1
  threshold_pct?: number;
}): boolean {
  // Threshold resolution:
  // - If BOTH sector_sigma (non-null) and k are usable → threshold = k * sector_sigma (σ-driven gate; ROADMAP criterion-aware).
  // - Otherwise → threshold = threshold_pct ?? 1 (existing default).
  const k = args.k ?? 1;
  const threshold = args.sector_sigma != null && args.sector_sigma > 0
    ? k * args.sector_sigma
    : (args.threshold_pct ?? 1);

  // Primary path: sector-relative (Phase 21).
  if (args.sector_relative_pct != null) {
    return args.sector_relative_pct > threshold;
  }
  // Fallback path: legacy SPY-alpha (used for outcomes pre-dating Phase 21 backfill,
  // or where /api/cron/relabel could not retrieve sector ETF prices).
  if (args.spy_return_pct == null) return false;
  return (args.ticker_return_pct - args.spy_return_pct) > threshold;
}
```

```typescript
  ticker_return_pct: number;
  spy_return_pct: number | null;
  sector_relative_pct?: number | null;
  sector_sigma?: number | null;   // optional: σ of sector ETF returns over the horizon

    : (args.threshold_pct ?? 1);
  if (args.sector_relative_pct != null) {

  if (args.spy_return_pct == null) return false;
```
