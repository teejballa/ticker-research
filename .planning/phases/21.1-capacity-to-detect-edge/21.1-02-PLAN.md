import { cached } from '@/lib/data/cache';  // or whichever import path the file already uses for Upstash
import yahooFinance from 'yahoo-finance2';

/**
 * Phase 21.1 D-14 — 60-day rolling sector ETF volatility.
 *
 * Returns the sample standard deviation of {etf}'s daily returns over the
 * 60 trading days ending on (or just before) `asOf`. Expressed as a decimal
 * (e.g., 0.012 = 1.2% per day).
 *
 * Used as `sector_sigma` argument to classifyHit for the σ-aware k=1 label
 * that becomes the PRIMARY engine label per CONTEXT D-14.
 *
 * Snapshot discipline: the returned σ is persisted on PriceOutcome.sector_sigma_60d
 * at compute time and never re-resolved (mirrors Phase 21 sector_etf snapshot pattern).
 *
 * @knowable_at  asOf  — uses only close-price history ending on `asOf`; no forward data
 */
export async function getSectorSigma60d(
  etf: string,
  asOf: Date,
): Promise<number | null> {
  const dateKey = asOf.toISOString().slice(0, 10);
  return cached(`sector_sigma_60d:${etf}:${dateKey}`, 24 * 60 * 60, async () => {
    try {
      const period2 = asOf;
      const period1 = new Date(asOf.getTime() - 100 * 24 * 60 * 60 * 1000); // ~100 calendar days → ≥60 trading days
      const result = await yahooFinance.historical(etf, { period1, period2, interval: '1d' });
      if (!result || result.length < 30) return null;
      // Sort ascending by date; take last 61 rows so we get 60 returns
      const closes = result
        .filter((r) => r.close != null && r.close > 0)
        .sort((a, b) => a.date.getTime() - b.date.getTime())
        .slice(-61)
        .map((r) => r.close);
      if (closes.length < 31) return null;
      const returns: number[] = [];
      for (let i = 1; i < closes.length; i++) {
        returns.push(closes[i] / closes[i - 1] - 1);
      }
      // Sample stdev
      const mean = returns.reduce((a, b) => a + b, 0) / returns.length;
      const variance = returns.reduce((acc, r) => acc + (r - mean) ** 2, 0) / (returns.length - 1);
      return Math.sqrt(variance);
    } catch (err) {
      console.warn(`[getSectorSigma60d] ${etf}@${dateKey} failed:`, err);
      return null;
    }
  });
}

/**
 * Phase 21.1 D-18 — z-score feature companions.
 *
 * Two transforms over base features:
 *   - tickerRolling60dZ: each base feature vs the same ticker's own 60-day history
 *   - crossSectionalZ:   each base feature within today's BACKFILL_UNIVERSE
 *
 * The engine's logistic moves from 12 features → 36 features:
 *   12 base + 12 ticker-rolling-z + 12 cross-sectional-z
 *
 * Both functions are pure. They take precomputed history / universe vectors;
 * the caller is responsible for assembling those slices PIT-correctly.
 */

/**
 * @knowable_at  history60d's most recent date (caller ensures no forward leakage)
 */
export function tickerRolling60dZ(value: number, history60d: number[]): number {
  if (history60d.length < 5) {
    console.warn(`[tickerRolling60dZ] insufficient history (n=${history60d.length}); returning 0`);
    return 0;
  }
  const mean = history60d.reduce((a, b) => a + b, 0) / history60d.length;
  const variance = history60d.reduce((acc, x) => acc + (x - mean) ** 2, 0) / (history60d.length - 1);
  const std = Math.sqrt(variance);
  if (std === 0) {
    console.warn(`[tickerRolling60dZ] zero variance; returning 0`);
    return 0;
  }
  return (value - mean) / std;
}

/**
 * @knowable_at  asOf (universeValues must be observed strictly before/at asOf)
 */
export function crossSectionalZ(value: number, universeValues: number[]): number {
  if (universeValues.length < 5) {
    console.warn(`[crossSectionalZ] universe too small (n=${universeValues.length}); returning 0`);
    return 0;
  }
  const mean = universeValues.reduce((a, b) => a + b, 0) / universeValues.length;
  const variance = universeValues.reduce((acc, x) => acc + (x - mean) ** 2, 0) / (universeValues.length - 1);
  const std = Math.sqrt(variance);
  if (std === 0) {
    console.warn(`[crossSectionalZ] zero variance; returning 0`);
    return 0;
  }
  return (value - mean) / std;
}

/**

export const FEATURE_NAMES_12 = FEATURE_NAMES;  // alias for clarity

export const FEATURE_NAMES_36: readonly string[] = [
  ...FEATURE_NAMES_12,
  ...FEATURE_NAMES_12.map((n) => `${n}_z60d`),
  ...FEATURE_NAMES_12.map((n) => `${n}_zxs`),
] as const;

if (FEATURE_NAMES_36.length !== 36) {
  throw new Error(`FEATURE_NAMES_36 must have 36 entries, got ${FEATURE_NAMES_36.length}`);

/**
 * Phase 21.1 D-18 — assemble the 36-feature vector for a snapshot.
 *
 * @param snapshot       the TechnicalSnapshot at decision time
 * @param history60d     a Record<feature_name, history_value[]> — last 60 trading days of base features for the SAME ticker
 * @param universe       a Record<feature_name, universe_values[]> — today's universe values (as-of `snapshot.scanned_at`) of the base features
 *
 * @knowable_at  snapshot.scanned_at  (caller ensures history/universe are PIT-correct)
 */
export function buildFeatureVector36(
  snapshot: TechnicalSnapshot,
  history60d: Record<string, number[]>,
  universe: Record<string, number[]>,
): number[] {
  const base = buildFeatureVector12(snapshot);  // existing function
  if (base.length !== 12) {
    throw new Error(`buildFeatureVector12 returned ${base.length} features, expected 12`);
  }
  const zRolling = FEATURE_NAMES_12.map((name, i) => {
    const hist = history60d[name] ?? [];
    return tickerRolling60dZ(base[i], hist);
  });
  const zCrossSec = FEATURE_NAMES_12.map((name, i) => {
    const u = universe[name] ?? [];
    return crossSectionalZ(base[i], u);
  });
  return [...base, ...zRolling, ...zCrossSec];
}

/**

 * Wave 4 wires this into the engine logistic; Wave 3 baselines use it too.
 */
export function priorPrecisionForN(n: number): number {
  if (n < 100) return 8;
  if (n < 500) return 4;
  return 1;
}

  // sector_sigma_60d snapshotted at compute time (D-17 anti-drift)
  const sector_sigma_60d = args.sector_etf
    ? await getSectorSigma60d(args.sector_etf, args.asOf)
    : null;

  // D-15: directional — beat sector by > 0%
  const is_directional_hit = args.sector_relative_pct != null
    ? args.sector_relative_pct > 0
    : null;

  // D-14 PRIMARY: σ-aware k=1
  const is_sigma_hit_k1 = sector_sigma_60d != null
    ? classifyHit({
        ticker_return_pct: args.ticker_return_pct,
        spy_return_pct: args.spy_return_pct,
        sector_relative_pct: args.sector_relative_pct,
        sector_sigma: sector_sigma_60d,
        k: 1,
      })
    : null;

  // D-16 TERTIARY: 1%-flat (existing behavior preserved for back-compat)
  const is_hit_flat1 = args.sector_relative_pct != null
    ? classifyHit({
        ticker_return_pct: args.ticker_return_pct,
        spy_return_pct: args.spy_return_pct,
        sector_relative_pct: args.sector_relative_pct,
        threshold_pct: 1,
      })
    : null;

  return { is_directional_hit, is_sigma_hit_k1, is_hit_flat1, sector_sigma_60d };
}
