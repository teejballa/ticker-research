```ts
      vec.push(v as number);

  return vec;
```

```ts
import {
  initLogisticState,
  predictLogistic,
  updateLogistic,
  buildFeatureVector36,
  FEATURE_NAMES_36,
  type LogisticState,
} from '@/lib/learning';
import { buildCanonicalFeatures, CANONICAL_FEATURE_NAMES, type CanonicalSnapshot } from './canonical-features';
import { informationCoefficient, categoricalLogLoss } from '@/lib/evaluation';

export type BaselineName = 'engine36' | 'canonical7';

export interface TrainingRow {
  // 36-feature path: rich snapshot + history + universe
  snapshot?: any;
  history60d?: Record<string, number[]>;
  universe?: Record<string, number[]>;
  // 7-feature path: pre-joined canonical features
  canonical?: CanonicalSnapshot;
  // labels
  hit: 0 | 1;            // is_sigma_hit_k1 from Wave 2 (PRIMARY label per D-14)
  forward_return: number; // sector_relative_pct (for IC)
}

export interface TrainOpts {
  priorPrecision: number;  // overrides initLogisticState default; per D-13/D-19
}

export interface ScoreResult {
  brier: number | null;
  log_loss: number | null;
  ic: number;
  n: number;
  predictions: number[];   // for bootstrap CIs
  returns: number[];
}

/**
 * Phase 21.1 D-11/D-12 — train a named baseline by iterating online Bayesian
 * updates on the training rows. PRIOR_PRECISION is overridable per D-13 ridge sweep.
 *
 * @knowable_at  max(row.snapshot.scanned_at for row in rows) — caller is responsible
 *               for using only PIT-correct rows; cv.ts purgedKFold ensures this in evaluation
 */
export function trainBaseline(
  name: BaselineName,
  rows: TrainingRow[],
  opts: TrainOpts = { priorPrecision: 1.0 },
): LogisticState {
  const featureNames = name === 'engine36' ? [...FEATURE_NAMES_36] : [...CANONICAL_FEATURE_NAMES];
  // Init with overridden precision: 1/var = precision → var = 1/precision
  let state: LogisticState = {
    intercept: 0,
    intercept_var: 1 / opts.priorPrecision,
    weights: featureNames.map(() => 0),
    weight_vars: featureNames.map(() => 1 / opts.priorPrecision),
    feature_names: featureNames,
  };
  for (const row of rows) {
    const x = name === 'engine36'
      ? buildFeatureVector36(row.snapshot, row.history60d ?? {}, row.universe ?? {})
      : buildCanonicalFeatures(row.canonical!);
    state = updateLogistic(state, x, row.hit);
  }
  return state;
}

/**
 * Phase 21.1 — score a trained baseline against eval rows, returning Brier + log-loss + IC.
 *
 * Brier is on the binary hit label (is_sigma_hit_k1 from Wave 2).
 * IC is Spearman correlation of predicted probability vs forward_return.
 * Log-loss is 3-way Buy/Hold/Sell — derived from the probability by:
 *   p_buy = p (the predicted hit probability)
 *   p_sell = 1 - p (complement)
 *   p_hold ≈ 0; treat the 3-way classification by thresholding at 0.5 → Buy/Sell binary.
 *
 * NOTE: For a binary classifier, log-loss simplifies to binary cross-entropy.
 * We materialize the 3-way labels at the cron level (Task 3); this scoring fn
 * accepts pre-computed 3-way labels via opts.
 */
export function scoreBaseline(
  state: LogisticState,
  rows: TrainingRow[],
  name: BaselineName = 'engine36',
): ScoreResult {
  if (rows.length === 0) {
    return { brier: null, log_loss: null, ic: NaN, n: 0, predictions: [], returns: [] };
  }
  const predictions: number[] = [];
  const returns: number[] = [];
  const hits: (0 | 1)[] = [];
  for (const row of rows) {
    const x = name === 'engine36'
      ? buildFeatureVector36(row.snapshot, row.history60d ?? {}, row.universe ?? {})
      : buildCanonicalFeatures(row.canonical!);
    const p = predictLogistic(state, x);
    predictions.push(p);
    returns.push(row.forward_return);
    hits.push(row.hit);
  }
  const brier = predictions.reduce((acc, p, i) => acc + (p - hits[i]) ** 2, 0) / predictions.length;
  const ic = informationCoefficient(predictions, returns, 'spearman');
  // Binary log-loss (binary cross-entropy)
  const eps = 1e-15;
  const log_loss = -predictions.reduce((acc, p, i) => {
    const pc = Math.min(Math.max(p, eps), 1 - eps);
    return acc + (hits[i] === 1 ? Math.log(pc) : Math.log(1 - pc));
  }, 0) / predictions.length;
  return { brier, log_loss, ic, n: predictions.length, predictions, returns };
}

/**
 * Phase 21.1 D-13 — ridge precision sweep over a grid; picks the value that
 * minimizes OOS Brier averaged across Purged K-Fold + Embargo splits.
 *
 * Returns the chosen precision + a sweep table for HYPERPARAMETERS.md (Wave 6).
 */
export function sweepPriorPrecision(
  name: BaselineName,
  rows: TrainingRow[],
  grid: number[],
  cvSplits: Array<{ train: number[]; test: number[] }>,
): { chosen_prior: number; best_brier: number; sweep_table: Array<{ prior: number; mean_oos_brier: number; std_oos_brier: number }> } {
  const sweep_table: Array<{ prior: number; mean_oos_brier: number; std_oos_brier: number }> = [];
  for (const prior of grid) {
    const foldBriers: number[] = [];
    for (const split of cvSplits) {
      const train = split.train.map((i) => rows[i]);
      const test = split.test.map((i) => rows[i]);
      const state = trainBaseline(name, train, { priorPrecision: prior });
      const score = scoreBaseline(state, test, name);
      if (score.brier != null) foldBriers.push(score.brier);
    }
    const mean = foldBriers.reduce((a, b) => a + b, 0) / Math.max(foldBriers.length, 1);
    const variance = foldBriers.reduce((acc, b) => acc + (b - mean) ** 2, 0) / Math.max(foldBriers.length - 1, 1);
    sweep_table.push({ prior, mean_oos_brier: mean, std_oos_brier: Math.sqrt(variance) });
  }
  const best = sweep_table.reduce((a, b) => (a.mean_oos_brier < b.mean_oos_brier ? a : b));
  return { chosen_prior: best.prior, best_brier: best.mean_oos_brier, sweep_table };
}
```

```ts
  }

  // ── Purged K-Fold + Embargo splits (CORE-ML-16)
  const cvSplits = purgedKFold(trainingRows, 5, { embargo_pct: 0.01, horizon_days: 7 });

  // ── Ridge precision sweep per baseline (D-13)
  const sweepE36 = sweepPriorPrecision('engine36', trainingRows, [1, 2, 4, 8, 16], cvSplits);
  const sweepC7 = sweepPriorPrecision('canonical7', trainingRows, [0.5, 1, 2, 4], cvSplits);

  // ── Train final models on full set
  const e36State = trainBaseline('engine36', trainingRows, { priorPrecision: sweepE36.chosen_prior });
  const c7State = trainBaseline('canonical7', trainingRows, { priorPrecision: sweepC7.chosen_prior });

  // ── Evaluate on last fold's test set (out-of-sample for headline)
  const lastFold = cvSplits[cvSplits.length - 1];
  const testRows = lastFold.test.map((i) => trainingRows[i]);
  const e36Score = scoreBaseline(e36State, testRows, 'engine36');
  const c7Score = scoreBaseline(c7State, testRows, 'canonical7');

  // ── BCa 95% CIs on each metric
  const e36BrierCI = bootstrapBCa(
    e36Score.predictions.map((p, i) => (p - (testRows[i].hit)) ** 2),
    (s) => s.reduce((a, b) => a + b, 0) / s.length,
    { nResamples: 10000 },
  );
  const c7BrierCI = bootstrapBCa(
    c7Score.predictions.map((p, i) => (p - (testRows[i].hit)) ** 2),
    (s) => s.reduce((a, b) => a + b, 0) / s.length,
    { nResamples: 10000 },
  );

  // ── Null baseline (predict in-sample base rate every time)
  const baseRate = trainingRows.reduce((acc, r) => acc + r.hit, 0) / trainingRows.length;
  const nullBrier = testRows.reduce((acc, r) => acc + (baseRate - r.hit) ** 2, 0) / testRows.length;

  // ── Persist as LearningEvent
  const payload = {
    n_train: trainingRows.length,
    n_test: testRows.length,
    cv_folds: cvSplits.length,
    base_rate: baseRate,
    null_brier: nullBrier,
    engine36: {
      chosen_prior: sweepE36.chosen_prior,
      brier: e36Score.brier,
      brier_ci_bca_95: [e36BrierCI.low, e36BrierCI.high],
      ic: e36Score.ic,
      log_loss: e36Score.log_loss,
      sweep_table: sweepE36.sweep_table,
    },
    canonical7: {
      chosen_prior: sweepC7.chosen_prior,
      brier: c7Score.brier,
      brier_ci_bca_95: [c7BrierCI.low, c7BrierCI.high],
      ic: c7Score.ic,
      log_loss: c7Score.log_loss,
      sweep_table: sweepC7.sweep_table,
    },
  };
  await prisma.learningEvent.create({
    data: {
      event_type: 'baseline_eval',
      delta: payload as any,
      message: `baseline-eval ${new Date().toISOString()}`,
    },
  });

  console.log('[cron:baseline-eval] complete', { ms: Date.now() - startedAt });
  return NextResponse.json({ ok: true, ...payload });
}
```
