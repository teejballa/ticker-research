**Step 1** — Create `src/app/api/cron/llm-ic/route.ts`:

```ts
import { NextResponse } from 'next/server';
import { prisma } from '@/lib/db';

export const runtime = 'nodejs';
export const dynamic = 'force-dynamic';
export const maxDuration = 300;

const HORIZONS = [7, 14, 30] as const;

/**
 * Phase 21.1 CORE-ML-20 — LLM direct evaluation cron.
 *
 * For each recent Report (last 60d):
 *   - ll_signed_score s ∈ [-1, +1] per D-08:
 *       Buy  → +confidence_pct/100
 *       Hold →  0
 *       Sell → -confidence_pct/100
 *   - score EVERY prediction (D-09 — no confidence filter)
 *   - persist one LLMEvaluation row per (report, horizon)
 *   - on subsequent runs, resolve forward_return + brier_buy_prob when horizon elapses
 */
export async function GET(request: Request) {
  const auth = request.headers.get('authorization');
  if (auth !== `Bearer ${process.env.CRON_SECRET}`) {
    return new NextResponse('Unauthorized', { status: 401 });
  }
  const startedAt = Date.now();

  // ── Step A: insert new prediction rows for recent reports
  const sixtyDaysAgo = new Date(Date.now() - 60 * 24 * 60 * 60 * 1000);
  const recentReports = await prisma.report.findMany({
    where: { created_at: { gte: sixtyDaysAgo } },
    select: {
      id: true, ticker: true, created_at: true,
      assessment: true, confidence_pct: true,
    },
  });

  let n_new = 0;
  for (const report of recentReports) {
    if (!report.assessment || report.confidence_pct == null) continue;
    const conf = report.confidence_pct / 100;
    const s = report.assessment === 'BUY' ? +conf
            : report.assessment === 'SELL' ? -conf
            : 0;
    for (const h of HORIZONS) {
      try {
        await prisma.lLMEvaluation.create({
          data: {
            ticker: report.ticker,
            report_id: report.id,
            horizon_days: h,
            ll_signed_score: s,
            ic_contribution: s,
            is_resolved: false,
          },
        });
        n_new++;
      } catch (err: any) {
        // unique violation = already inserted on a prior run — skip silently
        if (err?.code !== 'P2002') throw err;
      }
    }
  }

  // ── Step B: resolve forward_return for any unresolved rows whose horizon has elapsed
  const unresolved = await prisma.lLMEvaluation.findMany({
    where: { is_resolved: false },
    include: { /* if there's a relation to Report — adapt to actual schema */ },
  });

  let n_resolved = 0;
  for (const ev of unresolved) {
    const eventDate = ev.recorded_at.getTime();
    const horizonMs = ev.horizon_days * 24 * 60 * 60 * 1000;
    if (Date.now() - eventDate < horizonMs) continue;  // horizon not yet elapsed

    // Try to locate PriceOutcome for (report_id, days_after = horizon_days)
    const outcome = ev.report_id
      ? await prisma.priceOutcome.findFirst({
          where: { report_id: ev.report_id, days_after: ev.horizon_days },
        })
      : null;
    if (!outcome) continue;

    const fwd = outcome.forward_return_sector_rel ?? outcome.pct_change;
    const hit = outcome.is_sigma_hit_k1;
    // Buy-prob from ll_signed_score: p_buy = (s + 1) / 2 (maps [-1,+1] → [0,1])
    const p_buy = (ev.ll_signed_score + 1) / 2;
    const brier_buy_prob = hit == null ? null : (p_buy - (hit ? 1 : 0)) ** 2;

    await prisma.lLMEvaluation.update({
      where: { id: ev.id },
      data: {
        forward_return: fwd,
        brier_buy_prob,
        is_resolved: true,
      },
    });
    n_resolved++;
  }

  const n_total_active = await prisma.lLMEvaluation.count({ where: { is_resolved: true } });

  console.log('[cron:llm-ic] complete', { ms: Date.now() - startedAt, n_new, n_resolved, n_total_active });
  return NextResponse.json({ ok: true, n_new, n_resolved, n_total_active });
}
```

```ts
import { informationCoefficient, bootstrapBCa, perSignalClassIC } from '@/lib/evaluation';
import { prisma } from '@/lib/db';

// ── Types
export interface RollingLLMICPoint { t: Date; ic: number; ci_low: number; ci_high: number; n: number }
export type LLMICVerdict = 'EDGE_DETECTED' | 'NEGATIVE_EDGE' | 'INCONCLUSIVE' | 'COLLECTING_DATA';
export interface RollingLLMICResult {
  series: RollingLLMICPoint[];
  current: { ic: number; ci_low: number; ci_high: number; n: number } | null;
  verdict: LLMICVerdict;
}

export interface BaselineComparisonRow {
  metric: 'brier' | 'ic' | 'log_loss';
  llm: { value: number; ci: [number, number]; q_value: number | null };
  logistic24: { value: number; ci: [number, number]; q_value: number | null };
  logisticCanonical: { value: number; ci: [number, number]; q_value: number | null };
  null: { value: number };
}
export interface BaselineComparisonResult {
  rows: BaselineComparisonRow[];
  n_trials_attempted: number;
  last_updated: Date | null;
}

export interface PerSignalIC { signal_class: string; ic: number; ci: [number, number]; n: number }

// ── getLLMICRolling — Surface 2 headline
/**
 * Phase 21.1 D-10 / CORE-ML-20 — rolling N-day LLM IC with BCa 95% CI band.
 *
 * Reads LLMEvaluation rows where is_resolved=true and recorded_at is in the
 * last `windowDays`. Computes one IC per day (sliding window).
 *
 * Per D-30, all numeric values flow through this reader — dashboards never
 * read raw LLMEvaluation rows directly.
 *
 * @knowable_at  recorded_at (every row carries its own recorded_at)
 */
export async function getLLMICRolling(windowDays: number = 30): Promise<RollingLLMICResult> {
  const cutoff = new Date(Date.now() - windowDays * 24 * 60 * 60 * 1000);
  const rows = await prisma.lLMEvaluation.findMany({
    where: { is_resolved: true, recorded_at: { gte: cutoff } },
    select: { recorded_at: true, ll_signed_score: true, forward_return: true, horizon_days: true },
    orderBy: { recorded_at: 'asc' },
  });

  if (rows.length < 30) {
    return { series: [], current: null, verdict: 'COLLECTING_DATA' };
  }

  const predictions = rows.map((r) => r.ll_signed_score);
  const returns = rows.map((r) => r.forward_return ?? 0);

  // Headline IC over entire window
  const ic = informationCoefficient(predictions, returns, 'spearman');

  // BCa CI via paired-resample of (pred, return) tuples
  const indices = rows.map((_, i) => i);
  const ci = bootstrapBCa(indices, (idx) => {
    const p = idx.map((i) => predictions[i]);
    const r = idx.map((i) => returns[i]);
    return informationCoefficient(p, r, 'spearman');
  }, { nResamples: 10000 });

  const current = { ic, ci_low: ci.low, ci_high: ci.high, n: rows.length };

  // Sliding-window series: one IC per day
  const series: RollingLLMICPoint[] = [];
  const sortedTs = rows.map((r) => r.recorded_at.getTime());
  const dayMs = 24 * 60 * 60 * 1000;
  const firstDay = Math.floor(sortedTs[0] / dayMs) * dayMs;
  const lastDay = Math.floor(sortedTs[sortedTs.length - 1] / dayMs) * dayMs;
  for (let day = firstDay; day <= lastDay; day += dayMs) {
    const windowStart = day - windowDays * dayMs;
    const inWindow = rows.filter((r) => r.recorded_at.getTime() > windowStart && r.recorded_at.getTime() <= day);
    if (inWindow.length < 30) continue;
    const p = inWindow.map((r) => r.ll_signed_score);
    const ret = inWindow.map((r) => r.forward_return ?? 0);
    const wIc = informationCoefficient(p, ret, 'spearman');
    const wCi = bootstrapBCa(inWindow.map((_, i) => i), (idx) => {
      const pp = idx.map((i) => p[i]);
      const rr = idx.map((i) => ret[i]);
      return informationCoefficient(pp, rr, 'spearman');
    }, { nResamples: 2000 });  // smaller B for the sparkline; main tile uses 10000
    series.push({ t: new Date(day), ic: wIc, ci_low: wCi.low, ci_high: wCi.high, n: inWindow.length });
  }

  // Verdict per UI-SPEC Surface 2
  let verdict: LLMICVerdict = 'INCONCLUSIVE';
  if (ci.low > 0) verdict = 'EDGE_DETECTED';
  else if (ci.high < 0) verdict = 'NEGATIVE_EDGE';

  return { series, current, verdict };
}

// ── getBaselineComparison — Surface 1 source
/**
 * Phase 21.1 D-28 / CORE-ML-21 — read most recent baseline_eval LearningEvent
 * and the most recent LLM IC + Brier from LLMEvaluation, returning the 3-way table.
 *
 * @knowable_at  most recent baseline_eval.occurred_at
 */
export async function getBaselineComparison(horizon: 3 | 7 | 14 | 30 = 7): Promise<BaselineComparisonResult> {
  const evt = await prisma.learningEvent.findFirst({
    where: { event_type: 'baseline_eval' },
    orderBy: { occurred_at: 'desc' },
  });
  if (!evt) {
    return { rows: [], n_trials_attempted: 0, last_updated: null };
  }
  const delta: any = evt.delta;
  const e36 = delta.engine36 ?? {};
  const c7 = delta.canonical7 ?? {};
  const nullBrier = delta.null_brier ?? 0.25;

  // LLM metrics at this horizon — aggregate over LLMEvaluation rows for the horizon
  const llmRows = await prisma.lLMEvaluation.findMany({
    where: { is_resolved: true, horizon_days: horizon },
    select: { ll_signed_score: true, forward_return: true, brier_buy_prob: true },
  });
  const llmIC = llmRows.length >= 10
    ? informationCoefficient(llmRows.map((r) => r.ll_signed_score), llmRows.map((r) => r.forward_return ?? 0), 'spearman')
    : NaN;
  const llmBrier = llmRows.length > 0
    ? llmRows.reduce((acc, r) => acc + (r.brier_buy_prob ?? 0), 0) / llmRows.length
    : NaN;

  const llmBrierCI = llmRows.length >= 30
    ? bootstrapBCa(llmRows.map((r) => r.brier_buy_prob ?? 0), (s) => s.reduce((a, b) => a + b, 0) / s.length, { nResamples: 5000 })
    : { low: NaN, high: NaN };

  // q-values — for v1, leave LLM q_value=null (we don't compare LLM with FDR against baselines in this rollup)
  // The dashboard cell-coloring rule uses CI overlap, not q-value, per UI-SPEC.

  const rows: BaselineComparisonRow[] = [
    {
      metric: 'brier',
      llm: { value: llmBrier, ci: [llmBrierCI.low, llmBrierCI.high], q_value: null },
      logistic24: { value: e36.brier ?? NaN, ci: (e36.brier_ci_bca_95 ?? [NaN, NaN]) as [number, number], q_value: null },
      logisticCanonical: { value: c7.brier ?? NaN, ci: (c7.brier_ci_bca_95 ?? [NaN, NaN]) as [number, number], q_value: null },
      null: { value: nullBrier },
    },
    {
      metric: 'ic',
      llm: { value: llmIC, ci: [NaN, NaN], q_value: null },
      logistic24: { value: e36.ic ?? NaN, ci: [NaN, NaN], q_value: null },

/**
 * Phase 21.1 D-29 — IC by signal class from LLMEvaluation joined to Report.
 * Returns one row per signal class observed in last 30 days.
 *
 * @knowable_at  most recent LLMEvaluation.recorded_at
 */
export async function getPerSignalIC(): Promise<PerSignalIC[]> {
  // For v1, since Report may not carry a signal_class label, fall back to aggregating
  // LearnedPattern rows (which DO carry signal_class) and reading rolling_ic_20d.
  // If the project's Report model carries a signal_class hint, prefer that route.
  const cells = await prisma.learnedPattern.findMany({
    where: { rolling_ic_20d: { not: null } },
    select: { signal_class: true, rolling_ic_20d: true, sample_size: true },
  });
  const byClass = new Map<string, number[]>();
  for (const c of cells) {
    if (c.rolling_ic_20d == null) continue;
    if (!byClass.has(c.signal_class)) byClass.set(c.signal_class, []);
    byClass.get(c.signal_class)!.push(c.rolling_ic_20d);
  }
  const out: PerSignalIC[] = [];
  for (const [signal_class, ics] of byClass) {
    if (ics.length < 5) continue;
    const mean = ics.reduce((a, b) => a + b, 0) / ics.length;
    const ci = bootstrapBCa(ics, (s) => s.reduce((a, b) => a + b, 0) / s.length, { nResamples: 5000 });
    out.push({ signal_class, ic: mean, ci: [ci.low, ci.high], n: ics.length });
  }
  out.sort((a, b) => Math.abs(b.ic) - Math.abs(a.ic));
  return out;
}
```

```tsx
export async function RollingLLMICTile() {
  const data = await getLLMICRolling(30);
  const badge = VERDICT_BADGES[data.verdict];
  const ic = data.current?.ic;
  const ciLow = data.current?.ci_low;
  const ciHigh = data.current?.ci_high;
  const n = data.current?.n;

  const fmt = (v: number) => (v >= 0 ? '+' : '') + v.toFixed(3);

  // SVG sparkline geometry
  const W = 600, H = 180;
  let pathLine = '', pathBand = '';
  if (data.series.length > 1) {
    const allLow = Math.min(0, ...data.series.map((p) => p.ci_low));

    const range = Math.max(allHigh - allLow, 1e-6);
    const xs = data.series.map((_, i) => (i / (data.series.length - 1)) * W);
    const yFn = (v: number) => H - ((v - allLow) / range) * H;
    pathLine = data.series.map((p, i) => `${i === 0 ? 'M' : 'L'}${xs[i]},${yFn(p.ic)}`).join(' ');

    const lower = data.series.slice().reverse().map((p, idx) => {
      const i = data.series.length - 1 - idx;
      return `L${xs[i]},${yFn(p.ci_low)}`;
    }).join(' ');
    pathBand = `${upper} ${lower} Z`;
  }

    <section
      className="rounded-lg border border-outline-variant bg-surface p-6"

        {ic != null ? fmt(ic) : '—'}

        {ic != null
```
